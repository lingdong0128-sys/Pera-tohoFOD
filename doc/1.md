要实现图片处理逻辑跟随屏幕等比变化（响应式缩放），我们需要引入一个**缩放因子（Scale Factor）**。这个因子基于“当前屏幕宽度”与“设计基准宽度”的比例计算得出。

主要修改集中在 `dynamic_loader.py` 文件中。

### 核心思路

1.  **定义基准分辨率**：假设你的游戏是基于 `1600` 宽度的设计（这是 `SimpleERAConsole` 中的默认值）。
2.  **计算缩放比例**：`scale_factor = 当前屏幕宽度 / 基准宽度`。
3.  **应用缩放**：在渲染图片（包括单图、行内图、叠加图）时，将所有的 尺寸（Width/Height）、偏移（Offset）、裁剪（Clip）都乘以这个比例。

#### 1. 修改 `InlineFragment` 类

我们需要让行内片段的计算和渲染支持缩放因子。

```python
# dynamic_loader.py 中的 InlineFragment 类

class InlineFragment:
    """行内片段 - 增强版，支持图片标记"""

    def __init__(self, text, color=(255, 255, 255), click_value=None,
                 is_image_mark=False, img_info=None, clip_pos=None, size=None):
        # ... (保持 __init__ 内容不变) ...
        self.text = str(text)
        self.color = color
        self.click_value = click_value
        self.is_image_mark = is_image_mark
        self.img_info = img_info
        self.clip_pos = clip_pos
        self.size = size
        self.width = 0
        self.height = 0

    # [修改] 增加 scale_factor 参数
    def calculate_width(self, font, scale_factor=1.0):
        """计算片段宽度"""
        if self.is_image_mark and self.img_info:
            # 获取原始目标宽度
            base_width = 0
            if self.size:
                base_width = self.size[0]
            else:
                base_width = self.img_info.get('original_width', 270)
            
            # 应用缩放
            self.width = int(base_width * scale_factor)
        else:
            # 普通文本宽度通常由字号决定，字号已经调整过的话这里不需要缩放
            # 如果字号没变但想强制缩放文本间距，也可以乘，但通常不必
            self.width = font.size(self.text)[0]

        return self.width

    # [修改] 增加 scale_factor 参数
    def render_image(self, scale_factor=1.0):
        """渲染图片（如果是图片标记）"""
        if not self.is_image_mark or not self.img_info:
            return None

        try:
            img_path = os.path.join(self.img_info.get('base_dir', './'),
                                    self.img_info.get('filename', ''))

            if not os.path.exists(img_path):
                return None

            image = pygame.image.load(img_path).convert_alpha()

            # 裁剪 (裁剪是基于原图的，不需要缩放坐标，除非clip_pos本身是相对的)
            # 这里假设 clip_pos 是基于原图像素的，所以先裁剪
            clip_x, clip_y = self.clip_pos if self.clip_pos else (0, 0)
            clip_width = self.img_info.get('width', 270) # CSV里的width通常是裁剪宽
            clip_height = self.img_info.get('height', 270)

            img_width, img_height = image.get_size()
            
            # 安全裁剪
            safe_clip_w = min(clip_width, img_width - clip_x)
            safe_clip_h = min(clip_height, img_height - clip_y)
            
            if safe_clip_w > 0 and safe_clip_h > 0:
                clipped_image = image.subsurface(pygame.Rect(clip_x, clip_y, safe_clip_w, safe_clip_h))
            else:
                return None

            # --- 缩放逻辑 ---
            target_w, target_h = 0, 0
            
            if self.size:
                # 如果指定了 size，先按 size 缩放
                target_w, target_h = self.size
            else:
                # 否则使用裁剪后的尺寸
                target_w, target_h = safe_clip_w, safe_clip_h

            # 应用全局屏幕缩放因子
            final_w = int(target_w * scale_factor)
            final_h = int(target_h * scale_factor)

            # 执行缩放 (使用 smoothscale 质量更好)
            if final_w > 0 and final_h > 0:
                clipped_image = pygame.transform.smoothscale(clipped_image, (final_w, final_h))

            return clipped_image

        except Exception as e:
            print(f"渲染行内图片失败: {e}")
            return None
```

#### 5. 修改 `_render_and_draw_image_stack` (图片叠加的缩放)

这是最重要的地方，处理立绘叠加的等比缩放。

```python
    def _render_and_draw_image_stack(self, screen, item, x, y):
        """渲染图片叠加 - 支持等比缩放"""
        metadata = item.metadata
        
        # 获取原始模板尺寸
        raw_template_width = metadata.get('template_width', self.screen_width - 20)
        raw_template_height = metadata.get('template_height', 300)
        
        # [修改] 应用缩放因子到模板尺寸
        template_width = int(raw_template_width * self.scale_factor) if raw_template_width < self.screen_width else self.screen_width - 20
        # 高度也进行缩放，保持比例
        template_height = int(raw_template_height * self.scale_factor)
        
        img_elements = metadata.get('img_elements', [])

        if not img_elements:
            self._draw_image_error(screen, x, y, template_height)
            return None

        try:
            # 创建透明画布
            final_surface = pygame.Surface((template_width, template_height), pygame.SRCALPHA)

            for element in img_elements:
                info = element['info']
                img_path = info.get('path')
                image = self._get_image_from_cache(img_path)
                if not image: continue

                # 1. 裁剪 (基于原图，不缩放坐标)
                csv_x = info.get('clip_x', 0)
                csv_y = info.get('clip_y', 0)
                csv_w = info.get('original_width', image.get_width())
                csv_h = info.get('original_height', image.get_height())
                
                # ... (裁剪逻辑不变，见原代码) ...
                if csv_w > 0 and csv_h > 0: # 简化的裁剪逻辑
                     image = image.subsurface(pygame.Rect(csv_x, csv_y, min(csv_w, image.get_width()-csv_x), min(csv_h, image.get_height()-csv_y)))

                # 2. 计算目标尺寸 (应用缩放)
                # 获取 CSV 或参数中定义的目标尺寸
                raw_target_w = info.get('target_width', image.get_width())
                raw_target_h = info.get('target_height', image.get_height())
                
                # [修改] 乘以缩放因子
                target_width = int(raw_target_w * self.scale_factor)
                target_height = int(raw_target_h * self.scale_factor)

                # 执行缩放
                if target_width > 0 and target_height > 0:
                    image = pygame.transform.smoothscale(image, (target_width, target_height))

                # 3. 计算偏移 (应用缩放)
                raw_offset_x = info.get('offset_x', 0)
                raw_offset_y = info.get('offset_y', 0)
                
                # [修改] 乘以缩放因子
                offset_x = int(raw_offset_x * self.scale_factor)
                offset_y = int(raw_offset_y * self.scale_factor)
                
                final_surface.blit(image, (offset_x, offset_y))

            # 绘制到屏幕
            screen.blit(final_surface, (x, y))
            
            # [修改] 更新点击区域尺寸
            if metadata.get('clickable'):
                rect = pygame.Rect(x, y, template_width, template_height)
                # ... (更新 active_clickable_regions 逻辑不变) ...

            return final_surface

        except Exception as e:
            print(f"渲染图片叠加失败: {e}")
            return None
```

### 总结

做了这些修改后：
1.  如果你在 `config.json` 把屏幕宽度改成 `3200`，所有图片会自动放大 2 倍。
2.  如果你改成 `800`，图片会自动缩小一半。
3.  CSV 里配置的 `x, y` 坐标和 `width, height` 都会被自动换算，无需手动修改 CSV。
